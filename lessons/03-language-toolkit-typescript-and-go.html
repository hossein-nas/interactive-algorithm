1. Arrays and slices

Plain English: an array in TypeScript is a growable list. In Go, the everyday type is a slice: a small window onto an underlying array, which also grows when you append.

The two classic array traps

TypeScript: fill([]) shares one inner array. Array(3).fill([]) puts the same inner list in all three slots. Use Array.from.

Go: a sub-slice shares memory with the original. Appending to the sub-slice can overwrite the original's data. When you need an independent copy (saving a path in backtracking is the big one), call copy explicitly.

2D arrays (grids, DP tables)

Watch both traps happen. The first version is the TypeScript one; switch to the second for Go.

function makeGrid(rows: number, cols: number, fillValue = 0): number[][] {
  return Array.from({ length: rows }, () => new Array<number>(cols).fill(fillValue));
}

const bad = Array(3).fill([] as number[]); // all three rows are the SAME array
bad[0].push(1);
console.log(bad); // [ [ 1 ], [ 1 ], [ 1 ] ]  <- surprise

const good = makeGrid(2, 3, 7);
good[0][1] = 9;
console.log(good); // [ [ 7, 9, 7 ], [ 7, 7, 7 ] ]

// A numeric sort needs a comparator. The default sort compares as text.
console.log([10, 9, 1].sort()); // [ 1, 10, 9 ]
console.log([10, 9, 1].sort((a, b) => a - b)); // [ 1, 9, 10 ]
func makeGrid(rows, cols, fill int) [][]int {
	grid := make([][]int, rows)
	for r := range grid {
		grid[r] = make([]int, cols) // each row is its own slice
		for c := range grid[r] {
			grid[r][c] = fill
		}
	}
	return grid
}

func exampleGridsAndSlices() {
	g := makeGrid(2, 3, 7)
	g[0][1] = 9
	fmt.Println(g) // [[7 9 7] [7 7 7]]

	// Sub-slice trap: b shares memory with a.
	a := []int{1, 2, 3, 4}
	b := a[:2]
	b = append(b, 99) // overwrites a[2]!
	fmt.Println(a, b) // [1 2 99 4] [1 2 99]

	// Safe copy.
	c := make([]int, 2)
	copy(c, a[:2])
	c = append(c, 100)
	fmt.Println(a, c) // [1 2 99 4] [1 2 100]
}

Key difference: a Go string is a read-only sequence of bytes. For plain English letters that is the same as characters. For anything with accents or emoji, convert to []rune first, or your len and indexing will be wrong.

function reverseString(s: string): string {
  return [...s].reverse().join("");
}

// Count letters a..z with a 26-slot array (faster and simpler than a Map).
function letterCounts(s: string): number[] {
  const counts = new Array<number>(26).fill(0);
  for (let i = 0; i < s.length; i++) counts[s.charCodeAt(i) - 97]++;
  return counts;
}

func reverseString(s string) string {
	r := []rune(s)
	for i, j := 0, len(r)-1; i < j; i, j = i+1, j-1 {
		r[i], r[j] = r[j], r[i]
	}
	return string(r)
}

func letterCounts(s string) [26]int {
	var counts [26]int // an array (not a slice) is comparable with == and usable as a map key
	for i := 0; i < len(s); i++ {
		counts[s[i]-'a']++
	}
	return counts
}

Keys that are arrays or structs. In TypeScript, a Map compares object keys by identity, so [1,2] and [1,2] are two different keys. Convert to a string key such as "1,2". In Go, you can use an array like [26]int or a struct of comparable fields directly as a key (this makes "group anagrams" easy), but not a slice.

function frequency(nums: number[]): Map<number, number> {
  const freq = new Map<number, number>();
  for (const x of nums) freq.set(x, (freq.get(x) ?? 0) + 1);
  return freq;
}

console.log(frequency([1, 2, 2, 3, 3, 3])); // Map(3) { 1 => 1, 2 => 2, 3 => 3 }

// Map.get may legitimately return 0, so test with `=== undefined`, not truthiness.
const m = new Map<string, number>([["zero", 0]]);
console.log(m.get("zero") ? "found" : "missing"); // missing   <- bug if you meant "found"
console.log(m.get("zero") !== undefined ? "found" : "missing"); // found

// Arrays as keys are compared by identity, so use a string key.
const groups = new Map<string, string[]>();
for (const w of ["eat", "tea", "tan"]) {

func frequency(nums []int) map[int]int {

	for _, x := range nums {

	}

func exampleMapRecipes() {
	fmt.Println(frequency([]int{1, 2, 2, 3, 3, 3})) // map[1:1 2:2 3:3]  (fmt prints maps sorted)

	// A missing key and a stored zero look the same; use the comma-ok form to tell them apart.
	m := map[string]int{"zero": 0}
	v, ok := m["zero"]
	w, ok2 := m["nope"]
	fmt.Println(v, ok, w, ok2) // 0 true 0 false

	// An array can be a map key: group anagrams without sorting strings.
	groups := make(map[[26]int][]string)
	for _, word := range []string{"eat", "tea", "tan"} {
		var key [26]int
		for i := 0; i < len(word); i++ {
			key[word[i]-'a']++
		}
		groups[key] = append(groups[key], word)
	}
	fmt.Println(len(groups)) // 2

	seen := map[int]struct{}{1: {}, 2: {}, 3: {}}
	_, has := seen[2]
	fmt.Println(has, len(seen)) // true 3
}

4. Stack, queue, and deque

Stack (last in, first out): both languages use an array/slice and work at the end.

Queue (first in, first out): do not use shift() in TypeScript (it is O(n)). Keep a head index and just move it forward. In Go, q = q[1:] is O(1), but a head index is clearer and avoids holding old memory.

// Stack
const stack: number[] = [];
stack.push(1);
stack.push(2);
console.log(stack.pop(), stack[stack.length - 1]); // 2 1

// Queue with a head index: O(1) per operation, no shifting.
class Queue<T> {
  private items: T[] = [];
  private head = 0;
  push(x: T): void {
    this.items.push(x);
  }
  shift(): T | undefined {
    if (this.head >= this.items.length) return undefined;
    return this.items[this.head++];
  }
  get size(): number {
    return this.items.length - this.head;
  }
}

const q = new Queue<string>();
q.push("a");
q.push("b");
console.log(q.shift(), q.size); // a 1
func exampleStackQueue() {
	// Stack
	stack := []int{}
	stack = append(stack, 1)
	stack = append(stack, 2)
	top := stack[len(stack)-1]
	stack = stack[:len(stack)-1]
	fmt.Println(top, stack) // 2 [1]

	// Queue: take from the front by moving the start of the slice.
	queue := []string{"a", "b"}
	front := queue[0]
	queue = queue[1:]
	fmt.Println(front, len(queue)) // a 1

	// Deque: container/list is a doubly linked list.
	dq := list.New()
	dq.PushBack(1)
	dq.PushFront(0)
	fmt.Println(dq.Front().Value, dq.Back().Value, dq.Len()) // 0 1 2
}

5. The heap (priority queue)

Plain English: a heap lets you add items and always remove the smallest (min-heap) or largest (max-heap) in O(log n). It is the tool behind "top K", "merge K sorted lists", "scheduling", and Dijkstra.

  • TypeScript has no built-in heap. You write a small one. It is worth practicing until you can type it in two minutes.
  • Go has container/heap. You supply five methods and Go does the sifting. It is verbose but standard.
class MinHeap<T> {
  private data: T[] = [];
  constructor(private less: (a: T, b: T) => boolean) {}

  get size(): number {
    return this.data.length;
  }
  peek(): T | undefined {
    return this.data[0];
  }

  push(item: T): void {
    this.data.push(item);
    let i = this.data.length - 1;
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (!this.less(this.data[i], this.data[parent])) break;
      [this.data[i], this.data[parent]] = [this.data[parent], this.data[i]];
      i = parent;
    }
  }

  pop(): T | undefined {
    if (this.data.length === 0) return undefined;
    const top = this.data[0];
    const last = this.data.pop()!;
    if (this.data.length > 0) {
      this.data[0] = last;
      let i = 0;
      for (;;) {
        const l = 2 * i + 1, r = 2 * i + 2;
        let smallest = i;
        if (l < this.data.length && this.less(this.data[l], this.data[smallest])) smallest = l;
        if (r < this.data.length && this.less(this.data[r], this.data[smallest])) smallest = r;
        if (smallest === i) break;
        [this.data[i], this.data[smallest]] = [this.data[smallest], this.data[i]];
        i = smallest;
      }
    }
    return top;
  }
}

const minHeap = new MinHeap<number>((a, b) => a < b);
[5, 1, 4, 2].forEach((x) => minHeap.push(x));
console.log(minHeap.pop(), minHeap.pop(), minHeap.peek()); // 1 2 4

// Max-heap: flip the comparison. Pairs: order by the first field.
const maxHeap = new MinHeap<[number, string]>((a, b) => a[0] > b[0]);
maxHeap.push([3, "c"]);
maxHeap.push([9, "z"]);
console.log(maxHeap.pop()); // [ 9, 'z' ]
// 1) A min-heap of ints: five methods.
type IntHeap []int

func (h IntHeap) Len() int           { return len(h) }
func (h IntHeap) Less(i, j int) bool { return h[i] < h[j] } // flip to > for a max-heap
func (h IntHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *IntHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *IntHeap) Pop() any {
	old := *h
	n := len(old)
	x := old[n-1]
	*h = old[:n-1]
	return x
}

// 2) A heap of pairs, ordered by dist. This is the Dijkstra shape.
type Item struct {
	node, dist int
}
type ItemHeap []Item

func (h ItemHeap) Len() int           { return len(h) }
func (h ItemHeap) Less(i, j int) bool { return h[i].dist < h[j].dist }
func (h ItemHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *ItemHeap) Push(x any)        { *h = append(*h, x.(Item)) }
func (h *ItemHeap) Pop() any {
	old := *h
	n := len(old)
	x := old[n-1]
	*h = old[:n-1]
	return x
}

func exampleHeaps() {
	h := &IntHeap{5, 1, 4}
	heap.Init(h) // O(n) heapify
	heap.Push(h, 2)
	fmt.Println(heap.Pop(h), heap.Pop(h), (*h)[0]) // 1 2 4

	ih := &ItemHeap{}
	heap.Push(ih, Item{node: 7, dist: 30})
	heap.Push(ih, Item{node: 3, dist: 10})
	fmt.Println(heap.Pop(ih).(Item)) // {3 10}
}

Go heap gotchas: call heap.Push(h, x) and heap.Pop(h) (the package functions), not h.Push or h.Pop (those are the raw methods the package uses). heap.Pop returns any, so you must type-assert (.(int)).

Two interview-relevant surprises:

  1. Midpoint overflow. (lo + hi) / 2 can overflow in languages with fixed-size ints. The safe form is lo + (hi - lo) / 2. In Go with 64-bit int it rarely matters but is still good style. JavaScript does not overflow here but its division is floating point, so always floor it.
  2. Negative modulo. -7 % 3 is -1 in both languages (not 2). If you wrap around an array (rotation, circular buffers) use the positive-modulo form.
function mid(lo: number, hi: number): number {
  return lo + Math.floor((hi - lo) / 2);
}
function positiveMod(a: number, b: number): number {
  return ((a % b) + b) % b;
}

7. Structs and linked structures used in problems

Linked lists, trees, and graphs appear everywhere. Here are the usual definitions, so you do not fumble them.

class ListNode {
  constructor(public val: number, public next: ListNode | null = null) {}
}
class TreeNode {
  constructor(
    public val: number,
    public left: TreeNode | null = null,
    public right: TreeNode | null = null,
  ) {}
}

function listFromArray(values: number[]): ListNode | null {
  const dummy = new ListNode(0);
  let tail = dummy;
  for (const v of values) {
    tail.next = new ListNode(v);
    tail = tail.next;
  }
  return dummy.next;
}
function listToArray(head: ListNode | null): number[] {
  const out: number[] = [];
  for (let n = head; n !== null; n = n.next) out.push(n.val);
  return out;
}

const list3 = listFromArray([1, 2, 3]);
console.log(listToArray(list3)); // [ 1, 2, 3 ]

type ListNode struct {
	Val  int
	Next *ListNode
}
type TreeNode struct {
	Val         int
	Left, Right *TreeNode

func listFromSlice(values []int) *ListNode {
	dummy := &ListNode{}
	tail := dummy
	for _, v := range values {
		tail.Next = &ListNode{Val: v}
		tail = tail.Next
	}
	return dummy.Next
}
func listToSlice(head *ListNode) []int {
	out := []int{}
	for n := head; n != nil; n = n.Next {
		out = append(out, n.Val)
	}
	return out
}

	fmt.Println(tree.Left.Val, tree.Right.Val) // 2 3

}

8. Sorting with custom rules

Many problems say "sort by this, then by that" (intervals by start, words by length, pairs by count).

const people: [string, number][] = [["bob", 30], ["amy", 30], ["cat", 25]];
// Oldest first; ties broken by name A to Z.
people.sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
console.log(people); // [ [ 'amy', 30 ], [ 'bob', 30 ], [ 'cat', 25 ] ]

// Always copy first if the caller's array must stay in its original order.
const original = [3, 1, 2];
const sorted = [...original].sort((a, b) => a - b);
console.log(original, sorted); // [ 3, 1, 2 ] [ 1, 2, 3 ]
func exampleCustomSort() {
	type person struct {
		name string
		age  int
	}
	people := []person{{"bob", 30}, {"amy", 30}, {"cat", 25}}
	sort.Slice(people, func(i, j int) bool {
		if people[i].age != people[j].age {
			return people[i].age > people[j].age // oldest first
		}
		return people[i].name < people[j].name // then name A to Z
	})
	fmt.Println(people) // [{amy 30} {bob 30} {cat 25}]
}

9. Recursion limits

  • JavaScript has a fixed call-stack size (roughly 10,000 frames, depending on the frame size). A recursive walk on a chain of 100,000 nodes will throw RangeError: Maximum call stack size exceeded. For deep inputs use an explicit stack (a loop with your own array).
  • Go stacks grow dynamically (up to 1 GB by default), so deep recursion is rarely a problem, but it still costs memory.

A good interview line: "Recursion is clearest here. If the input could be a very long chain, I would convert it to an explicit stack."

2 · From shift() to a head index

Convert a "queue using shift()" into the version with a head index. Why is the original O(n²) overall?

Answer

Keep the array and an integer head. push appends; shift returns items[head] and increments head; the size is items.length - head. The original is O(n²) overall because each shift() moves every remaining item one slot left, which is (n−1) + (n−2) + … + 1 moves for n shifts. Lesson 02, Problem A animates exactly that.

3 · Letter counts for banana

Count letter frequencies for "banana" using a 26-slot array in both languages.

Answer

4 · What heap.Pop returns

What does heap.Pop(h) return in Go, and how do you turn it into an int?

Answer

heap.Pop(h) returns a value of type any (the empty interface). Turn it into an int with a type assertion: x := heap.Pop(h).(int). For a heap of structs it is heap.Pop(h).(Item). Call the package function heap.Pop(h), not the method h.Pop(), which skips the sifting.

5 · Two ways a sub-slice corrupts data

Show two ways a Go sub-slice can corrupt data, and how you would prevent it.

Answer
  1. Appending to a sub-slice with spare capacity writes into the original's backing array: b := a[:2]; b = append(b, 99) overwrites a[2]. Prevent it with an explicit copy, or limit the capacity with a full slice expression a[:2:2], which forces append to allocate.
  2. Saving a slice that you keep modifying, such as appending the current path to the results in backtracking and then changing the path. Every saved result shares the same memory and changes with it. Save append([]int(nil), path...) or a copy instead.
